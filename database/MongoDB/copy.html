#创建网络
docker network create mynetwork

docker network ls


#创建复制集 < --replSet myset --port 27017> 这部分必须写后边
docker run -d --net mynetwork --name mongo1 -v /Users/press/Data/mymongo/data1:/data/db -p 27017:27017   mongo:4 --replSet myset --port 27017

docker run -d --net mynetwork --name mongo2 -v /Users/press/Data/mymongo/data2:/data/db -p 27018:27018   mongo:4 --replSet myset --port 27018

docker run -d --net mynetwork --name mongo3 -v /Users/press/Data/mymongo/data3:/data/db -p 27019:27019   mongo:4 --replSet myset --port 27019

#进入一个节点
docker exec -it mongo1 mongo

#初始化复制集
rs.initiate(
	{
		_id: "myset",
		members: [
			{_id: 0, host: "mongo1:27017"},
			{_id: 1, host: "mongo2:27018"},
			{_id: 2, host: "mongo3:27019"},
		]
	}
)

#查看状态
rs.status()
